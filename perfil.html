} // Manejador de peticiones GET - Enruta todas las llamadas del sistema (Páginas HTML externas) function doGet(e) { try { if (!e || !e.parameter || !e.parameter.action) { return responderJSON({ success: false, message: "Parámetros o acción ausentes en la petición GET." }); } } if (!params.action) { return responderJSON({ success: false, message: "No se especificó ninguna acción en el POST." }); } return responderJSON(procesarPeticion(params)); } catch (error) { return responderJSON({ success: false, message: "Error interno en ejecución POST: " + error.toString() });
